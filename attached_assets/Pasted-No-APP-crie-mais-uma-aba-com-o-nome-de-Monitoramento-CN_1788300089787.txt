No APP crie mais uma aba com o nome de Monitoramento CNL e seja criativo com o menu para monitar as cheias do rio e estações.
const express = require("express");
const axios = require("axios");
const cheerio = require("cheerio");
const path = require("path");

const app = express();
const PORT = process.env.PORT || 3000;

app.use(express.json());
app.use(express.static(path.join(__dirname, "public")));

const CEMADEN_URL =
  "https://resources.cemaden.gov.br/graficos/interativo/grafico_CEMADEN.php?idpcd=6622&uf=MG";

app.get("/api/rio-bananeiras", async (req, res) => {
  try {
    const response = await axios.get(CEMADEN_URL, {
      timeout: 15000,
      headers: {
        "User-Agent": "Mozilla/5.0"
      }
    });

    const $ = cheerio.load(response.data);

    const texto = $("body").text().replace(/\s+/g, " ").trim();

    res.json({
      sucesso: true,
      estacao: {
        id: "6622",
        codigo: "311830410H",
        nome: "Rio Bananeiras",
        municipio: "Conselheiro Lafaiete",
        uf: "MG"
      },
      dados_brutos: texto,
      atualizado_em: new Date().toISOString(),
      fonte: CEMADEN_URL
    });

  } catch (error) {
    console.error("Erro CEMADEN:", error.message);

    res.status(500).json({
      sucesso: false,
      erro: "Não foi possível consultar o CEMADEN",
      mensagem: error.message
    });
  }
});

app.get("/api/status", (req, res) => {
  res.json({
    online: true,
    sistema: "Monitoramento Rio Bananeiras",
    horario: new Date().toISOString()
  });
});

app.listen(PORT, "0.0.0.0", () => {
  console.log(`Servidor rodando na porta ${PORT}`);
});

⚠️ Porém, tem uma limitação importante
Esse primeiro código não deve ser usado ainda como versão final.

A página do CEMADEN é dinâmica. Portanto, simplesmente baixar o HTML com axios pode retornar a estrutura da página sem os valores que o navegador carrega posteriormente.

Para fazer o que você realmente quer — nível do Rio Bananeiras + chuva + todas as estações + vento + umidade + temperatura + mapa + atualização automática — precisamos descobrir as requisições de dados que o site faz por trás.

A arquitetura final seria:

CEMADEN
   │
   ├── Rio Bananeiras 6622
   ├── Pluviômetros
   └── Estações hidrológicas
          │
          ▼
       BACKEND
        Replit
          │
          ├── Banco de dados
          ├── Histórico
          └── Alertas
          │
          ▼
       DASHBOARD
          │
          ├── 🌊 Rio
          ├── 🌧️ Chuva
          ├── 🌡️ Temperatura
          ├── 💧 Umidade
          ├── 💨 Vento
          ├── 🗺️ Mapa
          └── 🚨 Alertas

Se você quiser fazer isso direito, eu posso pesquisar agora o endpoint que o resources.cemaden.gov.br usa por trás dessa página idpcd=6622 e montar o código já puxando o valor real do nível do Rio Bananeiras, em vez de apenas mostrar o HTML da página.

# 2. Configura datas automáticas em tempo real (Últimas 48 horas)
hoje = datetime.date.today()
dois_dias_atras = hoje - datetime.timedelta(days=2)
data_fim = hoje.strftime('%Y-%m-%d')
data_inicio = dois_dias_atras.strftime('%Y-%m-%d')

print(f"📡 Buscando focos de incêndio ativos entre {data_inicio} e {data_fim}...")

# 3. Limite Territorial Oficial de Ouro Branco - MG (Dados extraídos via código)
# Buscamos na base global de municípiosSimplificados pelo nome correto
municipios = ee.FeatureCollection("FAO/GAUL/2015/level2")
ouro_branco = municipios.filter(ee.Filter.eq('ADM2_NAME', 'Ouro Branco')).filter(ee.Filter.eq('ADM1_NAME', 'Minas Gerais'))

# 4. Coleta dados de focos de calor do satélite MODIS (NASA) dentro do limite real
focos_incendio = (ee.ImageCollection('MODIS/006/MOD14A1')
                  .filterDate(data_inicio, data_fim) 
                  .filterBounds(ouro_branco))

if focos_incendio.size().getInfo() == 0:
    print("ℹ️ Nenhuma imagem de satélite processada para Ouro Branco nas últimas 48 horas.")
else:
    ultima_imagem = focos_incendio.mosaic()
    mascara_fogo = ultima_imagem.select('FireMask').gte(7) # Filtra apenas fogo confirmado

    # Transforma pixels em pontos vetoriais dentro do polígono da cidade
    pontos_fogo = mascara_fogo.reduceToVectors(
        geometry=ouro_branco,
        scale=1000,
        geometryType='point',
        labelProperty='fogo'
    )

    resultados = pontos_fogo.getInfo()
    print("\n--- 🔥 RELATÓRIO OFICIAL DE OURO BRANCO ---")
    
    features = resultados.get('features', [])
    if len(features) == 0:
        print("✅ Excelente! Nenhum foco de incêndio ativo detectado dentro do município.")
    else:
        print(f"⚠️ Alerta: Detectados {len(features)} focos de calor ativos dentro de Ouro Branco!")
        for feat in features:
            coords = feat['geometry']['coordinates']
            print(f"📍 Coordenadas do foco: Longitude {coords[0]}, Latitude {coords[1]}")
